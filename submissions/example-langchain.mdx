## Overview

This integration lets LangChain agents generate speech using Cartesia's TTS API. Use it when your agent needs to produce audio output as part of a conversational or agentic workflow.

## Prerequisites

- A Cartesia API key ([get one here](https://play.cartesia.ai))
- Python 3.9+
- LangChain installed

## Installation

```bash
pip install langchain cartesia
```

## Quick start

<Tabs>
  <Tab title="Python">
```python
from cartesia import Cartesia
from langchain.tools import tool

client = Cartesia(api_key="your-api-key")

@tool
def speak(text: str) -> str:
    """Generate speech audio from text using Cartesia TTS."""
    audio = client.tts.bytes(
        model_id="sonic-2024-10-25",
        transcript=text,
        voice_id="a0e99841-438c-4a64-b679-ae501e7d6091",
        output_format={
            "container": "wav",
            "encoding": "pcm_f32le",
            "sample_rate": 44100,
        },
    )
    with open("output.wav", "wb") as f:
        f.write(audio)
    return "Audio saved to output.wav"

result = speak.invoke("Welcome to Cartesia.")
print(result)
# Output: Audio saved to output.wav
```

  <Tab title="JavaScript">
```javascript
import Cartesia from "@cartesia/cartesia-js";
import { tool } from "@langchain/core/tools";
import { z } from "zod";
import { writeFileSync } from "fs";

const client = new Cartesia({ apiKey: "your-api-key" });

const speak = tool(
  async ({ text }) => {
    const response = await client.tts.bytes({
      modelId: "sonic-2024-10-25",
      transcript: text,
      voiceId: "a0e99841-438c-4a64-b679-ae501e7d6091",
      outputFormat: {
        container: "wav",
        encoding: "pcm_f32le",
        sampleRate: 44100,
      },
    });
    writeFileSync("output.wav", Buffer.from(response));
    return "Audio saved to output.wav";
  },
  {
    name: "speak",
    description: "Generate speech audio from text using Cartesia TTS.",
    schema: z.object({ text: z.string() }),
  }
);

const result = await speak.invoke({ text: "Welcome to Cartesia." });
console.log(result);
// Output: Audio saved to output.wav
```
